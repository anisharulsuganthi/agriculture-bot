import os
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
import traceback

# [SECRET REMOVED 2026-09-25] The Version-1 file hard-coded the sender account and
# fell back to a real Gmail app password. Both moved to SENDER_EMAIL /
# EMAIL_APP_PASSWORD in backend/.env. Rotate the app password - it was exposed
# here and in git history.
SENDER_EMAIL = os.getenv("SENDER_EMAIL", "")
# For Gmail, you need an 'App Password' if 2-Step Verification is on.
SENDER_PASSWORD = os.getenv("EMAIL_APP_PASSWORD", "")

def send_email_alert(to_email: str, message_body: str):
    """
    Sends an email alert using SMTP (Gmail).
    """
    try:
        msg = MIMEMultipart()
        msg['From'] = SENDER_EMAIL
        msg['To'] = to_email
        msg['Subject'] = "🌱 Smart Farm Climate & Disease Alert"
        
        msg.attach(MIMEText(message_body, 'plain'))
        
        # Connect to Gmail SMTP Server
        server = smtplib.SMTP('smtp.gmail.com', 587)
        server.starttls()
        server.login(SENDER_EMAIL, SENDER_PASSWORD)
        text = msg.as_string()
        server.sendmail(SENDER_EMAIL, to_email, text)
        server.quit()
        
        return {"status": "success", "message": "Email sent successfully!"}
    except Exception as e:
        print(f"Failed to send Email via SMTP: {e}")
        traceback.print_exc()
        return {"status": "error", "message": str(e)}

def format_alert_message(session_name, crop_type, recommendations):
    """
    Formats the climate recommendation into a readable text format.
    """
    lines = [f"🌱 Smart Farm Alert for {crop_type} ({session_name})\n"]
    
    watering = recommendations.get('watering', {})
    if watering:
        lines.append(f"💧 Water: {watering.get('recommendation')}")
        lines.append(f"   Reason: {watering.get('reason')}\n")
        
    fertilizing = recommendations.get('fertilizing', {})
    if fertilizing:
        lines.append(f"🌿 Fertilize: {fertilizing.get('recommendation')}")
        lines.append(f"   Type: {fertilizing.get('fertilizer_type')}")
        lines.append(f"   Reason: {fertilizing.get('reason')}\n")
        
    disease = recommendations.get('disease_risk', {})
    if disease and disease.get('risk_level') in ['Medium', 'High']:
        lines.append(f"⚠️ Disease Risk ({disease.get('risk_level')}): {disease.get('action')}")
        lines.append(f"   Reason: {disease.get('message')}\n")
        
    return "\n".join(lines)
